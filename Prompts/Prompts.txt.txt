======= Prompt 1 =======
Genera un proyecto en Java usando Swing que muestre una interfaz gráfica simple.

El programa debe simular un personaje en pantalla con botones para interactuar.

Debe tener una ventana con:
- Un área donde se muestre texto de acciones
- Botones para ejecutar acciones

El código debe estar completo y ser ejecutable.
Usa buenas prácticas y estructura clara.

===== Respuesta =====


import javax.swing.*;
import java.awt.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

// Clase principal
public class PersonajeApp {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            new VentanaPrincipal();
        });
    }
}

// Clase de la ventana principal
class VentanaPrincipal extends JFrame {

    private JTextArea areaTexto;
    private Personaje personaje;

    public VentanaPrincipal() {
        personaje = new Personaje("Héroe");

        setTitle("Simulador de Personaje");
        setSize(400, 300);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        initComponentes();

        setVisible(true);
    }

    private void initComponentes() {
        setLayout(new BorderLayout());

        // Área de texto
        areaTexto = new JTextArea();
        areaTexto.setEditable(false);
        areaTexto.setFont(new Font("Arial", Font.PLAIN, 14));
        JScrollPane scrollPane = new JScrollPane(areaTexto);

        add(scrollPane, BorderLayout.CENTER);

        // Panel de botones
        JPanel panelBotones = new JPanel();
        panelBotones.setLayout(new GridLayout(1, 3));

        JButton btnAtacar = new JButton("Atacar");
        JButton btnDefender = new JButton("Defender");
        JButton btnDescansar = new JButton("Descansar");

        panelBotones.add(btnAtacar);
        panelBotones.add(btnDefender);
        panelBotones.add(btnDescansar);

        add(panelBotones, BorderLayout.SOUTH);

        // Eventos
        btnAtacar.addActionListener(e -> mostrarAccion(personaje.atacar()));
        btnDefender.addActionListener(e -> mostrarAccion(personaje.defender()));
        btnDescansar.addActionListener(e -> mostrarAccion(personaje.descansar()));
    }

    private void mostrarAccion(String accion) {
        areaTexto.append(accion + "\n");
    }
}

// Clase modelo Personaje
class Personaje {
    private String nombre;
    private int energia;

    public Personaje(String nombre) {
        this.nombre = nombre;
        this.energia = 100;
    }

    public String atacar() {
        if (energia >= 10) {
            energia -= 10;
            return nombre + " ataca! Energía restante: " + energia;
        } else {
            return nombre + " está demasiado cansado para atacar.";
        }
    }

    public String defender() {
        if (energia >= 5) {
            energia -= 5;
            return nombre + " se defiende. Energía restante: " + energia;
        } else {
            return nombre + " no tiene suficiente energía para defenderse.";
        }
    }

    public String descansar() {
        energia += 15;
        if (energia > 100) energia = 100;
        return nombre + " descansa. Energía recuperada: " + energia;
    }
}

        // Área de texto
        areaTexto = new JTextArea();

        areaTexto.setFont(new Font("Arial", Font.PLAIN, 14));

        panel.add(btnCorrer);
        panel.add(btnSaltar);
        panel.add(btnEjecutar);

        add(panel, BorderLayout.SOUTH);

        btnCaminar.addActionListener(e -> {
            personaje.setMovimiento(new Caminar());
            mostrar("Estrategia cambiada a CAMINAR");

        btnCorrer.addActionListener(e -> {
            personaje.setMovimiento(new Correr());
            mostrar("Estrategia cambiada a CORRER");
        });

        btnSaltar.addActionListener(e -> {
            personaje.setMovimiento(new Saltar());
            mostrar("Estrategia cambiada a SALTAR");
        });

        btnEjecutar.addActionListener(e -> mostrar(personaje.ejecutarMovimiento()));
    }

    private void mostrar(String texto) {
        areaTexto.append(texto + "\n");
    }
}

import javax.swing.*;
import java.awt.*;

public class PersonajeApp {

    }

// ===== STRATEGY =====
interface Movimiento {
    String mover(String nombre);
}

class Caminar implements Movimiento {
    public String mover(String nombre) {
        return nombre + " está caminando 🚶";
    }
}

class Correr implements Movimiento {
    public String mover(String nombre) {

    }

class Saltar implements Movimiento {

    }

// ===== TEMPLATE METHOD =====
abstract class AccionPersonaje {

    // Template Method
    public final String realizarAccion(String nombre) {
        StringBuilder sb = new StringBuilder();
        sb.append(iniciar(nombre)).append("\n");
        sb.append(ejecutarAccion(nombre)).append("\n");
        sb.append(finalizar(nombre));
        return sb.toString();
    }

    protected String iniciar(String nombre) {
        return nombre + " se prepara...";
    }

    protected abstract String ejecutarAccion(String nombre);

    protected String finalizar(String nombre) {
        return nombre + " termina la acción.";
    }
}

class AtacarAccion extends AccionPersonaje {
    protected String ejecutarAccion(String nombre) {
        return nombre + " lanza un ataque poderoso ⚔️";
    }
}

class DefenderAccion extends AccionPersonaje {
    protected String ejecutarAccion(String nombre) {
        return nombre + " levanta un escudo 🛡️";
    }
}

// ===== MODELO =====
class Personaje {
    private String nombre;
    private Movimiento movimiento;

    public Personaje(String nombre) {
        this.nombre = nombre;
        this.movimiento = new Caminar();
    }

    public void setMovimiento(Movimiento movimiento) {
        this.movimiento = movimiento;
    }

    public String ejecutarMovimiento() {
        return movimiento.mover(nombre);
    }

    public String ejecutarAccion(AccionPersonaje accion) {
        return accion.realizarAccion(nombre);
    }
}
